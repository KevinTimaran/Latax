        \begin{block}{Interpretación Geométrica}
            \scriptsize
            \begin{itemize}
                \item La trayectoria muestra \textbf{cómo se desplaza} $(x^*, y^*)$ al variar $z$: desde $(0.94, -0.43)$ en $z=-2.0$ km hasta $(1.42, -0.76)$ en $z=0.0$ km.
                \vspace{0.1cm}
                \item La variación es \textbf{suave y continua}, confirmando estabilidad de la solución.
                \vspace{0.1cm}
                \item La región mínima es la zona donde $A_{pred,i} \approx A_{obs,i}$ de manera más consistente.
            \end{itemize}
        \end{block}

        \vspace{0.2cm}

        \begin{exampleblock}{Relación con la Fuente Real}
            \scriptsize
            \begin{itemize}
                \item En $z = -1.10$ km la trayectoria pasa por $(1.21, -0.79)$ km — \textbf{muy cerca de la fuente real} $(1.2, -0.8)$ km.
                \vspace{0.1cm}
                \item La visualización \textbf{confirma gráficamente} lo encontrado por Gauss-Newton.
            \end{itemize}
        \end{exampleblock}
