% 8.2. Ejemplo numérico con valores Gauss-Newton
%=============================================

\begin{frame}{Ejemplo Numérico del Procedimiento de Cálculo}

    \begin{columns}[T]

        \column{0.46\textwidth}

        \begin{block}{Mecánica de Evaluación}
            \scriptsize
            Para cada iteración $k$, el algoritmo evalúa tres pasos por cada sensor:
            \begin{enumerate}
                \item \textbf{Geometría:} Calcula la distancia euclidiana 3D al sensor.
                \item \textbf{Física:} Evalúa la atenuación teórica ($A_{pred}$) con $A_0^*$ estimado.
                \item \textbf{Estadística:} Obtiene el residuo individual.
            \end{enumerate}
        \end{block}

        \vspace{0.1cm}

        \begin{block}{Acumulación del Costo}
            \scriptsize
            Este procedimiento se repite para los 12 sensores. La sumatoria de residuos cuadrados define el error global $E(m)$, que Gauss-Newton minimiza iterativamente.
        \end{block}

        \vspace{-0.10cm}

        \begin{exampleblock}{Solución Óptima}
            \centering
            \scriptsize
            $m^* = (1.1995,\ {-0.8012},\ {-1.1024})$ km \\
            $A_0^* = 10.0142$
        \end{exampleblock}

        \column{0.50\textwidth}

        \begin{exampleblock}{Traza de Ejecución Real (Sensor 9)}
            \tiny
            Evaluación en $m^* = (1.1995,\ {-0.8012},\ {-1.1024})$ \\
            Datos de $S_9$: $(0.5,\ 0.5,\ 0.0)$,\quad $Az_{obs} = 0.852201$

            \vspace{0.15cm}
            \textbf{Paso 1: Distancia Euclidiana ($R_9$)}
            \[
                R_9 = \sqrt{(0.5-1.1995)^2 + (0.5-(-0.8012))^2 + (0.0-(-1.1024))^2}
            \]
            \[
                R_9 = \sqrt{(-0.6995)^2 + (1.3012)^2 + (1.1024)^2} \approx 1.8433
            \]

            \vspace{0.1cm}
            \textbf{Paso 2: Amplitud Predicha ($A_{pred,9}$)}
            \[
                A_{pred,9} = 10.0142 \cdot \frac{e^{-1.8433}}{1.8433} \approx 0.859991
            \]

            \vspace{0.1cm}
            \textbf{Paso 3: Residuo Individual}
            \[
                \text{Residuo} = A_{obs,9} - A_{pred,9}
            \]
            \[
                \text{Residuo} = 0.852201 - 0.859991 = -0.007790
            \]
        \end{exampleblock}

    \end{columns}

\end{frame}
