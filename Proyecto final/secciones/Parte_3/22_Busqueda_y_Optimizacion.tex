\begin{frame}{Estrategia de Búsqueda y Optimización}

    \begin{columns}[T]

        \column{0.45\textwidth}

        \begin{block}{Método Principal: Gauss-Newton}
            \scriptsize
            \begin{itemize}
                \item Algoritmo iterativo de \textbf{mínimos cuadrados no lineales}.
                \vspace{0.1cm}
                \item Linealiza el modelo en cada iteración mediante el \textbf{jacobiano G}.
                \vspace{0.1cm}
                \item Estima simultáneamente las \textbf{4 incógnitas}: $x_{\text{real}},\ y_{\text{real}},\ z_{\text{real}},\ A_0$.
                \vspace{0.1cm}
                \item Convergió en \textbf{8 iteraciones} con tolerancia $10^{-6}$.
            \end{itemize}
        \end{block}

        \vspace{0.2cm}

        \begin{exampleblock}{Esquema Iterativo}
            \centering
            \scriptsize
            \[
                \Delta m = (G^T G)^{-1} G^T \Delta A_z
            \]
            \[
                m^{(k+1)} = m^{(k)} + \Delta m
            \]
        \end{exampleblock}

        \column{0.50\textwidth}

        \begin{block}{Acotación del Dominio de Búsqueda}
            \centering
            \scriptsize
            \vspace{0.1cm}
            \begin{tabular}{@{}lcc@{}}
                \toprule
                \textbf{Variable} & \textbf{Rango [km]} & \textbf{Justificación} \\ \midrule
                $x$ & $[-5,\ 5]$ & Cobertura horizontal \\[0.15cm]
                $y$ & $[-5,\ 5]$ & Cobertura horizontal \\[0.15cm]
                $z$ & $[-3,\ 1]$ & Profundidad del hipocentro \\[0.15cm]
                $A_0$ & $[0,\ 50]$ & Amplitud física positiva \\ \bottomrule
            \end{tabular}
        \end{block}

        \vspace{0.2cm}

        \begin{block}{Espacio de Búsqueda}
            \centering
            \scriptsize
            El vector de parámetros es:
            \[
                m = [x_{\text{real}},\ y_{\text{real}},\ z_{\text{real}},\ A_0]^T \in \mathbb{R}^4
            \]
        \end{block}

    \end{columns}

    \vspace{0.1cm}
    \begin{center}
        \footnotesize
        \textbf{Gauss-Newton linealiza iterativamente el modelo no lineal de atenuación.}
    \end{center}

\end{frame}
