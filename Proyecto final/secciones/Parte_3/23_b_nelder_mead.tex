\begin{frame}{Método de Optimización Principal: Gauss-Newton}

    \begin{columns}[T]

        %===========================================
        % COLUMNA IZQUIERDA: Fundamento matemático
        %===========================================
        \column{0.48\textwidth}

        \begin{block}{Fundamento Matemático}
            \scriptsize
            El modelo de atenuación es \textbf{no lineal} respecto a las coordenadas. Gauss-Newton lo linealiza en cada iteración mediante una aproximación de Taylor de primer orden:
            \[
                A_z \approx A_z(m^k) + G \cdot \Delta m
            \]
            donde $G$ es la \textbf{matriz jacobiana} $M \times 4$.
        \end{block}

        \vspace{0.15cm}

        \begin{exampleblock}{Ecuaciones del Método}
            \scriptsize
            \textbf{Actualización óptima:}
            \[
                \Delta m = (G^T G)^{-1} G^T \Delta A_z
            \]
            \textbf{Iteración:}
            \[
                m^{(k+1)} = m^{(k)} + \Delta m
            \]
            \textbf{Criterio de parada:} $\| \Delta m \| < 10^{-6}$
        \end{exampleblock}

        \begin{block}{Robustez del Método}
            \scriptsize
            Convergió en \textbf{8 iteraciones} con precisión $> 99.7\%$.
        \end{block}
